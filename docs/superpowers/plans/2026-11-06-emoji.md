# 表情(emoji)Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 评论框和后台编辑器有同一个表情选择器,所有页面里的表情统一显示成 Noto 样式。

**Architecture:** 数据里仍存 Unicode 字符。`scripts/build_emoji.py` 从 Noto 仓库挑出要打包的 SVG,并生成两份前端数据(选择器用的 `emoji-data.js`、替换用的 `emoji-keys.js`)。`emoji.js` 在渲染好的页面里把已打包的表情字符换成 `<img alt="😀">`;`emoji-picker.js` 是共用的选择器,评论框通过 `data-emoji-for` 属性接入,Vditor 通过一个自定义工具栏按钮接入。

**Tech Stack:** Python 标准库(构建脚本)、原生 JS、`node --test`、pytest、Vditor 4.0.1(已内置在 `app/static/vendor/vditor`)。

设计文档:`docs/superpowers/specs/2026-11-06-comments-redesign-design.md` 第 8 节。与评论改造计划(`2026-11-06-comments-redesign.md`)相互独立:Task 4 的评论框接入需要评论改造的 Task 4 已完成(依赖 `_comments.html`);其余任务不依赖它。

## Global Constraints

- 数据里只存 Unicode 字符,不存图片地址或短代码;评论 Markdown 渲染与 `sanitize_html` 不改。
- 统一样式用 Noto Emoji 的 **SVG 图片**(Apache 2.0),**不用** Noto 字体(Safari 对其彩色字体格式支持不可靠)。
- 只替换已打包的表情;国旗、肤色变体和 ZWJ 组合不打包,保持系统样式。
- 没有 JS 或图片缺失时退回系统表情;`alt` 保留原字符,复制出来仍是表情。
- 静态文件放 `app/static/emoji/noto/`,文件名是去掉 `FE0F` 的小写十六进制码点用 `-` 连接(例如 `1f600.svg`、`2764.svg`、`31-20e3.svg`)。附 Apache 2.0 `LICENSE` 和 `NOTICE`。
- 打包总体积预算 ≤ 8 MB(`build_emoji.py --max-bytes` 默认值);超出就缩小范围,不是放宽预算。
- 编辑区(Vditor)里的表情默认是系统样式;编辑区字体实验(Task 6)是可选项,独立可回退。
- `localStorage` 读写一律包 try/catch,页面在它不可用时也必须正常工作。
- 下载来的第三方文件视为不可信数据:每份放进各自新建的空目录,脚本在项目目录里运行,以路径参数传入,Python 用 `-I` 运行。
- 新增界面文字直接写中文。
- 每个任务结束时必须跑 `source .venv/bin/activate && python -m pytest tests -q && node --test tests/*.test.js`,全绿再提交。

def make_noto(tmp_path, names):
    noto = tmp_path / 'noto'
    (noto / 'svg').mkdir(parents=True)
    for name in names:
        (noto / 'svg' / name).write_text(f'<svg id="{name}"/>')
    (noto / 'LICENSE').write_text('Apache License fixture')
    return noto


def run_build(tmp_path, names, **kwargs):

    cldr = tmp_path / 'zh.xml'

    out = tmp_path / 'out'

```python
"""Build the emoji set the site ships: Noto SVGs plus the data the scripts read.

Usage (all three inputs are downloaded separately, see docs/superpowers/plans/2026-11-06-emoji.md):

    python -I scripts/build_emoji.py --noto DIR --emoji-test FILE --cldr-zh FILE

Standard library only. Output goes to app/static (--out to change):
    emoji/noto/<key>.svg, LICENSE, NOTICE   the images
    emoji-keys.js                           which emoji can be swapped for an image
    emoji-data.js                           the picker's groups, names and keywords
"""
import argparse
import json
import re
import shutil
import sys
import xml.etree.ElementTree as ET
from pathlib import Path

ZWJ = '‍'
VS16 = '️'
SKIN_TONES = {chr(cp) for cp in range(0x1F3FB, 0x1F400)}
SKIPPED_GROUPS = {'Component', 'Flags'}
GROUP_LABELS = {
    'Smileys & Emotion': '表情',
    'People & Body': '人物',
    'Animals & Nature': '动物与自然',
    'Food & Drink': '食物',
    'Travel & Places': '旅行',
    'Activities': '活动',
    'Objects': '物品',
    'Symbols': '符号',
}
DEFAULT_MAX_BYTES = 8_000_000
NOTICE = """\
The emoji images in this directory come from Noto Emoji
(https://github.com/googlefonts/noto-emoji), Copyright Google Inc.,
licensed under the Apache License, Version 2.0. See LICENSE.

The images are copied unmodified and renamed after their code points.
Run scripts/build_emoji.py to regenerate them.
"""
_LINE = re.compile(
    r'^(?P<cps>[0-9A-F ]+?)\s*;\s*(?P<status>[\w-]+)\s*#\s*\S+\s+E[\d.]+\s+(?P<name>.+)$')


def strip_selector(chars):
    return chars.replace(VS16, '')


def key_of(chars):
    return '-'.join(f'{ord(c):x}' for c in strip_selector(chars))


def parse_emoji_test(text):
    """Fully-qualified emoji worth offering: (chars, group, name)."""
    group = None
    entries = []
    for line in text.splitlines():
        if line.startswith('# group:'):
            group = line.split(':', 1)[1].strip()
            continue
        match = _LINE.match(line)
        if not match or match['status'] != 'fully-qualified':
            continue
        if group in SKIPPED_GROUPS or group not in GROUP_LABELS:
            continue
        chars = ''.join(chr(int(cp, 16)) for cp in match['cps'].split())
        if ZWJ in chars or any(c in SKIN_TONES for c in chars):
            continue
        entries.append((chars, group, match['name'].strip()))
    return entries


def parse_cldr(xml_text):
    """{key: (short name, [keywords])} from a CLDR annotations file."""
    names, keywords = {}, {}
    for node in ET.fromstring(xml_text).iter('annotation'):
        key = key_of(node.get('cp', ''))
        text = (node.text or '').strip()
        if node.get('type') == 'tts':
            names[key] = text
        else:
            keywords[key] = [word.strip() for word in text.split('|') if word.strip()]
    return {key: (names.get(key), keywords.get(key, [])) for key in set(names) | set(keywords)}


def find_svg(noto_dir, chars):
    codepoints = [f'{ord(c):04x}' if ord(c) < 0x10000 else f'{ord(c):x}' for c in chars]
    plain = [f'{ord(c):04x}' if ord(c) < 0x10000 else f'{ord(c):x}'
             for c in strip_selector(chars)]
    for parts in (codepoints, plain):
        for name in ('_'.join(parts), '_'.join(p.lstrip('0') or '0' for p in parts)):
            path = Path(noto_dir) / 'svg' / f'emoji_u{name}.svg'
            if path.is_file():
                return path
    return None


def build(noto_dir, emoji_test, cldr_zh, out_dir, max_bytes=DEFAULT_MAX_BYTES):
    entries = parse_emoji_test(Path(emoji_test).read_text(encoding='utf-8'))
    annotations = parse_cldr(Path(cldr_zh).read_text(encoding='utf-8'))

    chosen, missing = [], []
    for chars, group, name in entries:
        svg = find_svg(noto_dir, chars)
        if svg is None:
            missing.append(chars)
        else:
            chosen.append((chars, group, name, svg))

    total = sum(svg.stat().st_size for *_, svg in chosen)
    if total > max_bytes:
        sys.exit(f'{total} bytes of SVG is over the {max_bytes} byte budget; '
                 'narrow the groups instead of raising the budget')

    out_dir = Path(out_dir)
    images = out_dir / 'emoji' / 'noto'
    if images.exists():
        shutil.rmtree(images)
    images.mkdir(parents=True)
    groups = {}
    for chars, group, name, svg in chosen:
        key = key_of(chars)
        shutil.copyfile(svg, images / f'{key}.svg')
        zh_name, words = annotations.get(key, (None, []))
        item = [chars, zh_name or name, ' '.join([*words, name.lower()])]
        groups.setdefault(group, []).append(item)

    shutil.copyfile(Path(noto_dir) / 'LICENSE', images / 'LICENSE')
    (images / 'NOTICE').write_text(NOTICE, encoding='utf-8')
    keys = ' '.join(key_of(chars) for chars, *_ in chosen)
    (out_dir / 'emoji-keys.js').write_text(
        f'window.FlogEmojiKeys = "{keys}";\n', encoding='utf-8')
    data = {'groups': [
        {'id': group.lower().replace(' & ', '-').replace(' ', '-'),
         'label': GROUP_LABELS[group], 'items': items}
        for group, items in groups.items()]}
    (out_dir / 'emoji-data.js').write_text(
        'window.FlogEmoji = ' + json.dumps(data, ensure_ascii=False, separators=(',', ':')) + ';\n',
        encoding='utf-8')
    return {'copied': len(chosen), 'missing': missing, 'bytes': total}


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__.split('\n')[0])
    parser.add_argument('--noto', required=True, type=Path, help='a checkout of googlefonts/noto-emoji')
    parser.add_argument('--emoji-test', required=True, type=Path)
    parser.add_argument('--cldr-zh', required=True, type=Path)
    parser.add_argument('--out', type=Path, default=Path(__file__).parents[1] / 'app' / 'static')
    parser.add_argument('--max-bytes', type=int, default=DEFAULT_MAX_BYTES)
    args = parser.parse_args(argv)
    result = build(args.noto, args.emoji_test, args.cldr_zh, args.out, args.max_bytes)
    print(f"copied {result['copied']} images, {result['bytes']} bytes")
    if result['missing']:
        print(f"{len(result['missing'])} emoji have no Noto image: {''.join(result['missing'])}")


if __name__ == '__main__':
    main()
```

```javascript
(() => {
  const SCOPES = '.article-body, .page-body, .post-comment__text';
  const SKIPPED = 'pre, code, script, style, textarea, .katex, .mermaid';
  const EMOJI = /\p{Regional_Indicator}{2}|[#*0-9]️?⃣|\p{Extended_Pictographic}(?:️|\p{Emoji_Modifier})?(?:‍\p{Extended_Pictographic}(?:️|\p{Emoji_Modifier})?)*/gu;

  const emojiKey = (chars) => Array.from(chars.replace(/️/g, ''))
    .map((c) => c.codePointAt(0).toString(16))
    .join('-');

  // Cut text into plain pieces and the emoji that have an image.
  const splitText = (text, keys) => {
    const pieces = [];
    let last = 0;
    for (const match of text.matchAll(EMOJI)) {
      const key = emojiKey(match[0]);
      if (!keys.has(key)) continue;
      if (match.index > last) pieces.push({ text: text.slice(last, match.index) });
      pieces.push({ emoji: match[0], key });
      last = match.index + match[0].length;
    }
    if (last < text.length) pieces.push({ text: text.slice(last) });
    return pieces;
  };

  if (typeof module !== 'undefined' && module.exports) {
    module.exports = { emojiKey, splitText };
  }
  if (typeof document === 'undefined') return;

  const base = document.currentScript?.dataset.base || '/static/emoji/noto/';
  const keys = new Set((window.FlogEmojiKeys || '').split(' ').filter(Boolean));
  if (!keys.size) return;

  const toImage = (piece) => {
    const image = document.createElement('img');
    image.className = 'emoji';
    image.alt = piece.emoji;
    image.draggable = false;
    image.src = `${base}${piece.key}.svg`;
    image.addEventListener('error', () => image.replaceWith(piece.emoji), { once: true });
    return image;
  };

  const replaceIn = (root) => {
    const walker = document.createTreeWalker(root, NodeFilter.SHOW_TEXT);
    const nodes = [];
    while (walker.nextNode()) nodes.push(walker.currentNode);
    nodes.forEach((node) => {
      if (node.parentElement.closest(SKIPPED)) return;
      const pieces = splitText(node.nodeValue, keys);
      if (pieces.length === 1 && pieces[0].text !== undefined) return;
      node.replaceWith(...pieces.map((piece) => (
        piece.text !== undefined ? piece.text : toImage(piece))));
    });
  };

  const run = () => document.querySelectorAll(SCOPES).forEach(replaceIn);
  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', run);
  else run();
})();
```

```javascript
(() => {
  const RECENT_KEY = 'flog-emoji-recent';
  const RECENT_MAX = 24;

  const searchGroups = (groups, query) => {
    const words = query.toLowerCase().split(/\s+/).filter(Boolean);
    if (!words.length) return [];
    const found = [];
    groups.forEach((group) => group.items.forEach((item) => {
      const haystack = `${item[1]} ${item[2]}`.toLowerCase();
      if (words.every((word) => haystack.includes(word))) found.push(item);
    }));
    return found;
  };

  const pushRecent = (list, chars, max = RECENT_MAX) => (
    [chars, ...list.filter((item) => item !== chars)].slice(0, max));

  const insertAtCursor = (field, text, makeEvent = (type) => new Event(type, { bubbles: true })) => {
    const start = field.selectionStart ?? field.value.length;
    const end = field.selectionEnd ?? start;
    field.setRangeText(text, start, end, 'end');
    field.focus();
    field.dispatchEvent(makeEvent('input'));
  };

  if (typeof module !== 'undefined' && module.exports) {
    module.exports = { searchGroups, pushRecent, insertAtCursor };
  }
  if (typeof document === 'undefined') return;

  const base = document.currentScript?.dataset.base || '/static/emoji/noto/';
  const keyOf = (chars) => Array.from(chars.replace(/️/g, ''))
    .map((c) => c.codePointAt(0).toString(16)).join('-');

  const readRecent = () => {
    try {
      const list = JSON.parse(localStorage.getItem(RECENT_KEY) || '[]');
      return Array.isArray(list) ? list.filter((item) => typeof item === 'string') : [];
    } catch (_error) {
      return [];
    }
  };
  const writeRecent = (list) => {
    try { localStorage.setItem(RECENT_KEY, JSON.stringify(list)); } catch (_error) { /* storage unavailable */ }
  };

  let panel = null;
  let opener = null;

  const close = () => {
    if (!panel) return;
    panel.remove();
    panel = null;
    document.removeEventListener('keydown', onKey, true);
    document.removeEventListener('pointerdown', onOutside, true);
    opener?.focus();
    opener = null;
  };
  const onKey = (event) => {
    if (event.key === 'Escape') { event.preventDefault(); close(); }
  };
  const onOutside = (event) => {
    if (panel && !panel.contains(event.target) && !opener?.contains(event.target)) close();
  };

  const makeItem = (item, onPick) => {
    const button = document.createElement('button');
    button.type = 'button';
    button.className = 'emoji-picker__item';
    button.title = item[1];
    const image = document.createElement('img');
    image.className = 'emoji';
    image.alt = item[0];
    image.loading = 'lazy';
    image.src = `${base}${keyOf(item[0])}.svg`;
    button.append(image);
    button.addEventListener('click', () => onPick(item[0]));
    return button;
  };

  const fill = (host, sections, onPick) => {
    host.replaceChildren();
    sections.forEach(({ label, items, id }) => {
      if (!items.length) return;
      const heading = document.createElement('h3');
      heading.textContent = label;
      if (id) heading.id = `emoji-group-${id}`;
      const grid = document.createElement('div');
      grid.className = 'emoji-picker__grid';
      items.forEach((item) => grid.append(makeItem(item, onPick)));
      host.append(heading, grid);
    });
    if (!host.children.length) {
      const empty = document.createElement('p');
      empty.className = 'emoji-picker__empty';
      empty.textContent = '没有找到匹配的表情';
      host.append(empty);
    }
  };

  const open = ({ anchor, onPick }) => {
    const groups = window.FlogEmoji?.groups;
    if (!groups) return;
    close();
    opener = anchor;

    panel = document.createElement('div');
    panel.className = 'emoji-picker';
    panel.setAttribute('role', 'dialog');
    panel.setAttribute('aria-label', '选择表情');

    const search = document.createElement('input');
    search.type = 'search';
    search.className = 'form-control emoji-picker__search';
    search.placeholder = '搜索表情';
    search.setAttribute('aria-label', '搜索表情');

    const tabs = document.createElement('div');
    tabs.className = 'emoji-picker__tabs';
    const list = document.createElement('div');
    list.className = 'emoji-picker__list';

    const byChars = new Map(groups.flatMap((g) => g.items).map((item) => [item[0], item]));
    const pick = (chars) => {
      writeRecent(pushRecent(readRecent(), chars));
      close();
      onPick(chars);
    };
    const showAll = () => {
      const recent = readRecent().map((chars) => byChars.get(chars)).filter(Boolean);
      fill(list, [{ label: '最近使用', items: recent, id: 'recent' }, ...groups], pick);
    };

    [{ label: '最近', id: 'recent' }, ...groups].forEach((group) => {
      const tab = document.createElement('button');
      tab.type = 'button';
      tab.textContent = group.label;
      tab.addEventListener('click', () => {
        search.value = '';
        showAll();
        list.querySelector(`#emoji-group-${group.id}`)?.scrollIntoView({ block: 'start' });
      });
      tabs.append(tab);
    });
    search.addEventListener('input', () => {
      const query = search.value.trim();
      if (!query) { showAll(); return; }
      fill(list, [{ label: '搜索结果', items: searchGroups(groups, query) }], pick);
    });

    showAll();
    panel.append(search, tabs, list);
    document.body.append(panel);

    const box = anchor.getBoundingClientRect();
    const width = Math.min(panel.offsetWidth || 340, window.innerWidth - 16);
    panel.style.top = `${window.scrollY + box.bottom + 6}px`;
    panel.style.left = `${Math.max(8, Math.min(window.scrollX + box.left, window.scrollX + window.innerWidth - width - 8))}px`;
    document.addEventListener('keydown', onKey, true);
    document.addEventListener('pointerdown', onOutside, true);
    search.focus();
  };

  window.FlogEmojiPicker = { open, close };

  document.addEventListener('click', (event) => {
    const trigger = event.target.closest('[data-emoji-for]');
    if (!trigger) return;
    const field = document.querySelector(trigger.dataset.emojiFor);
    if (!field) return;
    event.preventDefault();
    open({ anchor: trigger, onPick: (chars) => insertAtCursor(field, chars) });
  });
})();
```
